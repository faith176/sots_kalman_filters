@startuml
title Process Flow of the Reliability-Enhanced CEP Framework

actor Researcher

participant Stream
participant Coordinator
participant EventStream
participant Reconstructor
participant Predictor
participant ConfidenceModel
participant CEPEngine

== Event Generation ==
Researcher -> Stream: startDataCollection()
Stream -> Coordinator: generateEvent()
Coordinator -> EventStream: publish(event, "observed")

== Reconstruction ==
EventStream -> Reconstructor: dispatch(event, "observed")

alt Event observed
    Reconstructor -> ConfidenceModel: computeConfidence(event)
    ConfidenceModel --> Reconstructor: confidence
    Reconstructor -> EventStream: publish(event + metadata, "reconstructed")
else Event missing (gap detected)
    Coordinator -> Reconstructor: triggerReconstruction()
    Reconstructor -> Predictor: predict()
    Predictor --> Reconstructor: predicted_value
    Reconstructor -> ConfidenceModel: computeConfidence(predicted_value)
    ConfidenceModel --> Reconstructor: confidence
    Reconstructor -> EventStream: publish(reconstructed_event, "reconstructed")
end

== CEP Processing ==
EventStream -> CEPEngine: dispatch(event, "reconstructed")
CEPEngine -> CEPEngine: matchPatterns()
CEPEngine --> Researcher: logDetectedEvents()
@enduml
