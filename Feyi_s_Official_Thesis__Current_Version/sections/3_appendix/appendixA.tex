\section{Kalman Filter}\label{sec:kalman-formulation}
The Kalman filter is a recursive method for estimating the state of a linear dynamical system \cite{kalman1960filtering,bishop2001introduction}. The system state at time step $k$ is represented by a state vector $\vec{x}_k$, with associated uncertainty represented by a covariance matrix $\Sigma_k$. The quantities $\vec{x}_{k|k-1}$ and $\Sigma_{k|k-1}$ represent the prior state and covariance, while $\vec{x}_{k|k}$ and $\Sigma_{k|k}$ represent the posterior state and covariance.

\paragraph{System model.}
The Kalman filter assumes a linear dynamical system:
\[
\vec{x}_k = \mathbf{F}_k \vec{x}_{k-1} + \mathbf{B}_k \vec{u}_{k-1} + \vec{w}_k,
\qquad
\vec{z}_k = \mathbf{H}_k \vec{x}_k + \vec{v}_k,
\]
where:
\begin{itemize}
    \item $\mathbf{F}_k$ is the state transition matrix,
    \item $\mathbf{B}_k$ is the control input matrix,
    \item $\vec{u}_{k-1}$ is the control input,
    \item $\mathbf{H}_k$ is the observation matrix,
    \item $\vec{w}_k \sim \mathcal{N}(0, \mathbf{Q}_k)$ is process noise,
    \item $\vec{v}_k \sim \mathcal{N}(0, \mathbf{R}_k)$ is measurement noise.
\end{itemize}

\paragraph{Prediction step.}
The prior estimate is obtained by propagating the previous posterior:
\[
\vec{x}_{k|k-1} = \mathbf{F}_k \vec{x}_{k-1} + \mathbf{B}_k \vec{u}_{k-1},
\qquad
\Sigma_{k|k-1} = \mathbf{F}_k \Sigma_{k-1} \mathbf{F}_k^{T} + \mathbf{Q}_k.
\]

\paragraph{Update step.}
Given a measurement $\vec{z}_k$, the innovation and its covariance are defined as:
\[
\vec{y}_k = \vec{z}_k - \mathbf{H}_k \vec{x}_{k|k-1},
\qquad
\mathbf{S}_k = \mathbf{H}_k \Sigma_{k|k-1} \mathbf{H}_k^{T} + \mathbf{R}_k.
\]

The Kalman gain is:
\[
\mathbf{K}_k = \Sigma_{k|k-1} \mathbf{H}_k^{T} \mathbf{S}_k^{-1}.
\]

The posterior update is:
\[
\vec{x}_{k|k} = \vec{x}_{k|k-1} + \mathbf{K}_k \vec{y}_k,
\]
\[
\Sigma_{k|k} = (\mathbf{I} - \mathbf{K}_k \mathbf{H}_k)\,\Sigma_{k|k-1}.
\]

\section{Normalized Innovation Squared (NIS)}
\label{sec:predictor-confidence}

The Normalized Innovation Squared (NIS) is a statistical consistency measure for Kalman filters \cite{bar2001estimation,simon2006optimal}. It evaluates whether the innovation is consistent with its expected covariance.

Using the innovation $\vec{y}_k$ and innovation covariance $\mathbf{S}_k$ defined above, the NIS statistic is:
\[
\mathrm{NIS}_k =
\vec{y}_k^{T} \mathbf{S}_k^{-1} \vec{y}_k.
\]


\section{CEP Pattern Definitions}\label{app:patterns}
This section provides the Esper EPL specifications for the complex event patterns used in the evaluation. Each pattern captures correlated behaviour across multiple constituents.

\begin{lstlisting}[language=sql, caption={Cross-signal CEP pattern definitions}, label={lst:patterns}]

-- Cross-Signal Agreement: signals remain aligned
@Name('CrossSignalAgreement') select a1,b1,c1,a2,b2,c2 from pattern [every (a1=Event(src='dt-1') and b1=Event(src='dt-2') and c1=Event(src='dt-3')) -> (a2=Event(src='dt-1') and b2=Event(src='dt-2') and c2=Event(src='dt-3'))] where a1.eventTs<a2.eventTs and abs(a1.value-b1.value)<1.2 and abs(a1.value-c1.value)<1.2 and abs(b1.value-c1.value)<1.2 and abs(a2.value-b2.value)<1.2 and abs(a2.value-c2.value)<1.2 and abs(b2.value-c2.value)<1.2;

-- Cross-Signal Divergence: signals separate
@Name('CrossSignalDivergence') select a1,b1,c1,a2,b2,c2 from pattern [every (a1=Event(src='dt-1') and b1=Event(src='dt-2') and c1=Event(src='dt-3')) -> (a2=Event(src='dt-1') and b2=Event(src='dt-2') and c2=Event(src='dt-3'))] where a1.eventTs<a2.eventTs and (abs(a1.value-b1.value)>1.5 or abs(a1.value-c1.value)>1.5 or abs(b1.value-c1.value)>1.5) and (abs(a2.value-b2.value)>1.5 or abs(a2.value-c2.value)>1.5 or abs(b2.value-c2.value)>1.5);

-- Cross-Signal Stability: relative differences remain stable
@Name('CrossSignalRelativeStable') select a1,b1,c1,a2,b2,c2 from pattern [every (a1=Event(src='dt-1') and b1=Event(src='dt-2') and c1=Event(src='dt-3')) -> (a2=Event(src='dt-1') and b2=Event(src='dt-2') and c2=Event(src='dt-3'))] where a1.eventTs<a2.eventTs and abs((a1.value-b1.value)-(a2.value-b2.value))<0.3 and abs((b1.value-c1.value)-(b2.value-c2.value))<0.3;

\end{lstlisting}
