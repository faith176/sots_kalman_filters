\textbf{R1} &
The system shall provide a mechanism to \textbf{identify missing
observations} by evaluating temporal expectations over event streams and
detecting when an expected event fails to occur within its tolerance
window. &
\secref{sec:scheduling-missing};

\citet{wasserkrug2006taxonomy}

\textbf{R2} &
The system shall provide an \textbf{explicit representation for inferred
observations} so that reconstructed events remain distinguishable from
observed events through metadata. &
\secref{sec:state-based-inference};
\citet{wasserkrug2006taxonomy},
\citet{cugola2015complex}
\\

\textbf{R3} &
The system shall support \textbf{uncertainty propagation} by attaching
confidence metadata to reconstructed events, enabling downstream event
processing to account for occurrence uncertainty without modifying
query semantics. &
\secref{sec:confidence-stream-augmentation};
\citet{cugola2015complex},
\chref{ch:literature-review},
\secref{sec:literature-review-key-takeaways}
\\

\textbf{R4} &
The system shall enable \textbf{continued operation under temporary data
unavailability} by supporting inference over missing observations rather
than suspending event-driven reasoning. &
\secref{sec:scheduling-missing},
\secref{sec:state-based-inference},
\secref{subsec:literature-review-results-reliability},
\secref{sec:literature-review-key-takeaways}
\\

\textbf{R5} &
The system shall preserve \textbf{compatibility with existing CEP
engines} by producing reconstructed events that conform to the same
structural and temporal semantics as observed events. &
\secref{sec:events-streams},
\secref{sec:confidence-stream-augmentation},
\secref{subsec:literature-review-results-standards}
