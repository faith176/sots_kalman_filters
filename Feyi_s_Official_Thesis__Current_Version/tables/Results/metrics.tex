\textbf{CEP Detection Accuracy} &
Precision, Recall, F1-score &
Accuracy of CEP event detection compared to the oracle stream:
\[
\begin{aligned}
\text{Precision} &= \tfrac{TP}{TP + FP}, \qquad
\text{Recall} = \tfrac{TP}{TP + FN} \\
\text{F1} &= 2 \cdot \tfrac{\text{Precision} \cdot \text{Recall}}{\text{Precision} + \text{Recall}}
\end{aligned}
\]
\\

\addlinespace[0.4em]
\textbf{Constituent Reliability} &
Observation Reliability &
Probability that a constituent produces observations when expected:
\[
R_c = \frac{E_{produced}}{E_{expected}}.
\]
\\

\textbf{Reconstruction Accuracy} &
Mean Absolute Error (MAE), Root Mean Squared Error (RMSE) &
Reconstruction error between predicted and observed values:
\[
\begin{aligned}
\text{MAE} &= \frac{1}{N} \sum_{i=1}^{N} | \hat{x}_i - x_i | \\
\text{RMSE} &= \sqrt{ \frac{1}{N} \sum_{i=1}^{N} (\hat{x}_i - x_i)^2 }
\end{aligned}
\]
\\


\addlinespace[0.4em]
\textbf{Lifecycle State Distribution} &
State Duration Ratio &
Fraction of execution time constituents operate in each lifecycle state:
\[
S_i = \frac{T_{state_i}}{T_{total}}
\]
